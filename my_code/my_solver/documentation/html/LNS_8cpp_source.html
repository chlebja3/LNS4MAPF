74 void LNS::solve()
75 {
76  // start measuring time
77  Clock clock;
78  clock.start();
79 
80  // calculate initial solution
81  bool found = find_initial_solution();
82 
83  if (!found) // TODO restarts
84  {
85  return;
86  }
87  assertm(found, "Could not find the initial solution");
88  assertm(solution.is_valid(instance), "Found invalid solution");
89 
90  // std::cout << "Initial solution found in " << initial_solution_time << " second." << std::endl;
91 
92  // initialize the constraint table (needed for randomwalk and intersection destroy operator)
93  if (settings.destroy_settings.type != DESTROY_TYPE::RANDOM)
94  {
95  initialize_constraint_table(solution.paths);
96  }
97 
98  // improve the solution
99  while (iteration_num < settings.max_iter && clock.get_current_time().first < settings.time_limit)
100  {
101  // check for visualization thread end
102  if (settings.sipp_settings.info_type == INFO_type::visualisation)
103  {
104  if (shared_data->is_end.load(std::memory_order_acquire))
105  {
106  break;
107  }
108  }
109 
110  iteration_num++;
111 
112  // time the iteration
113  Clock iteration_clock;
114  iteration_clock.start();
115 
116  // copy solution
117  Solution perturbed_sol = solution;
118 
119  // destroy operator
120  destroy_operator.apply(perturbed_sol);
121 
122  // repair operator
123  repair_operator.apply(perturbed_sol);
124 
125  // accept only feasible solutions
126  bool accepted = false;
127  int improvement = 0;
128  if (!perturbed_sol.feasible)
129  {
130  discard_solution(perturbed_sol, solution);
131  std::cout << "Discarded infeasible solution." << std::endl;
132  }
133  else
134  {
135  // accept only improving solutions
136  perturbed_sol.calculate_cost(instance);
137  improvement = solution.sum_of_delays - perturbed_sol.sum_of_delays;
138 
139  // find out which destroy strategy was used
140  auto destroy_strategy_index_help = magic_enum::enum_index<DESTROY_TYPE>(last_destroy_strategy);
141  int destroy_strategy_index = -1;
142  if (destroy_strategy_index_help.has_value())
143  {
144  destroy_strategy_index = destroy_strategy_index_help.value();
145  }
146  assertm(destroy_strategy_index >= 0 && destroy_strategy_index < static_cast<int>(magic_enum::enum_count<DESTROY_TYPE>()),
147  "Wrong destroy strategy index.");
148 
149  if (improvement <= 0)
150  {
151  // update the weights
152  if (settings.destroy_settings.type == DESTROY_TYPE::ADAPTIVE)
153  {
154  destroy_weights[destroy_strategy_index] = (1 - decay_factor) * destroy_weights[destroy_strategy_index];
155  }
156  discard_solution(perturbed_sol, solution);
157  // std::cout << "Discarded worse solution, difference: " << perturbed_sol.sum_of_costs - solution.sum_of_costs << std::endl;
158  }
159  else
160  {
161  // update the weights
162  if (settings.destroy_settings.type == DESTROY_TYPE::ADAPTIVE)
163  {
164  destroy_weights[destroy_strategy_index] =
165  reaction_factor * static_cast<double>(improvement) / static_cast<double>(settings.destroy_settings.size) +
166  (1 - reaction_factor) * destroy_weights[destroy_strategy_index];
167  }
168  accepted = true;
169  // remember the perturbed solution
170  // std::cout << "Solution improved by: " << solution.sum_of_delays - perturbed_sol.sum_of_delays << std::endl;
171  solution = std::move(perturbed_sol);
172  }
173  }
174  auto [iteration_time_wall, iteration_time_cpu] = iteration_clock.end();
175 
176  // construct LNS iteration info and add to the shared data
177  if (settings.sipp_settings.info_type == INFO_type::visualisation)
178  {
179  // get the pointer to the original perturbed solution (which might have been moved)
180  Solution* sol = &solution;
181  if (!accepted)
182  {
183  sol = &perturbed_sol;
184  }
185  LNSIterationInfo lns_info(iteration_num, accepted, improvement, sipp_info, *sol);
186  sipp_info.clear();
187  shared_data->update_lns_info(lns_info);
188  }
189  // log the iteration
190  else if (settings.sipp_settings.info_type == INFO_type::experiment)
191  {
192  log.bsf_solution_cost.push_back(solution.sum_of_costs);
193  log.bsf_makespan.push_back(solution.makespan);
194  log.used_operator.push_back(last_destroy_strategy);
195  log.iteration_time_wall.push_back(iteration_time_wall);
196  log.iteration_time_cpu.push_back(iteration_time_cpu);
197  }
198  }

232 auto LNS::PrioritizedPlanning() -> Solution
233 {
234  Solution sol;
235  sol.paths.resize(instance.get_num_of_agents(), TimePointPath());
236  const int agent_num = instance.get_num_of_agents();
237 
238  // create a sequence from 0 to agent_num
239  std::vector<int> priorities(agent_num);
240  std::iota(priorities.begin(), priorities.end(), 0);
241 
242  // shuffle the sequence to get random priorities
243  std::shuffle(priorities.begin(), priorities.end(), rnd_generator);
244 
245  // initialize iter info
246  if (settings.sipp_settings.info_type == INFO_type::visualisation)
247  {
248  sipp_info.clear();
249  sipp_info.resize(agent_num);
250  }
251 
252  // plan path for each agent according to its priority
253  assertm(already_planned.empty(), "Running Prioritized Planning when some agents were already planned.");
254  for (int i = 0; i < agent_num; i++)
255  {
256  // get agent with priority i
257  int agent_id = priorities[i];
258 
259  // plan path for agent
260  TimePointPath tp_path = planner->plan(agent_id, already_planned);
261 
262  // get the visualization info
263  if (settings.sipp_settings.info_type == INFO_type::visualisation)
264  {
265  sipp_info[agent_id] = std::move(planner->iter_info);
266  planner->iter_info.clear();
267  }
268 
269  // check validity
270  if (tp_path.empty())
271  {
272  // std::cout << "Unable to find initial solution by prioritized planning. " << std::endl;
273  sol.feasible = false;
274  return sol;
275  }
276 
277  // check validity
278  assertm(instance.check_timepointpath_validity(tp_path), "SIPP planned an invalid timepointpath.");
279 
280  // update the safe interval table
281  planner->safe_interval_table.add_constraints(tp_path);
282 
283  // add to the solution
284  sol.paths[agent_id] = tp_path;
285 
286  // add to the already planned agents
287  already_planned.insert(agent_id);
288  }
289  // add priorities to the visualization info
290  if (settings.sipp_settings.info_type == INFO_type::visualisation)
291  {
292  sol.priorities = std::move(priorities);
293  }
294  return sol;
295 }

313 void LNS::destroy_randomwalk(Solution& sol) const
314 {
315  last_destroy_strategy = DESTROY_TYPE::RANDOMWALK;
316 
317  assertm(constraint_table_initialized, "Constraint table is not initialized.");
318  assertm(settings.destroy_settings.size >= 0 && settings.destroy_settings.size <= instance.get_num_of_agents(),
319  "Invalid neighborhood size.");
320  // if neighborhood size the same as number of agents or more, return all agents
321  if (settings.destroy_settings.size >= instance.get_num_of_agents())
322  {
323  destroy_random(sol);
324  return;
325  }
326 
327  // the tabu list
328  static std::unordered_set<int> tabu_list;
329 
330  // find the most delayed agent that is not on the tabu list
331  assertm(static_cast<int>(sol.delays.size()) == instance.get_num_of_agents(),
332  "The length of delay list must be the same as the numebr of agents.");
333 
334  int max_delay = 0;
335  int most_delayed = -1;
336  int latest_time = 0;
337  for (int i = 0; i < instance.get_num_of_agents(); i++)
338  {
339  if (sol.paths[i].back().interval.t_min > latest_time)
340  {
341  latest_time = sol.paths[i].back().interval.t_min;
342  }
343 
344  // skip agents in tabu list
345  if (tabu_list.find(i) != tabu_list.end())
346  {
347  continue;
348  }
349 
350  if (sol.delays[i] > max_delay)
351  {
352  max_delay = sol.delays[i];
353  most_delayed = i;
354  }
355  }
356 
357  // check, whether to reset tabu list
358  if (max_delay == 0)
359  {
360  // reset the tabu list and try again
361  if (!tabu_list.empty())
362  {
363  tabu_list.clear();
364  destroy_randomwalk(sol);
365  return;
366  }
367 
368  // select random agents
369  destroy_random(sol);
370  return;
371  }
372 
373  assertm(max_delay > 0, "Max delay is invalid.");
374  assertm(most_delayed >= 0 && most_delayed < instance.get_num_of_agents() && sol.delays[most_delayed] == max_delay,
375  "Invalid id of most delayed agent");
376 
377  std::unordered_set<int> chosen = {most_delayed};
378  int upperbound = sol.paths[most_delayed].back().interval.t_min - 1;
379  assertm(upperbound > 0, "Upperbound is too small.");
380 
381  // perform the random walk, maximum 10 iterations
382  for (int i = 0; i < 10; i++)
383  {
384  // choose a random agent
385  std::uniform_int_distribution<int> dist(0, chosen.size() - 1);
386  int idx = dist(rnd_generator);
387  assertm(idx >= 0 && idx < static_cast<int>(chosen.size()), "Index out of bounds.");
388  // iterator to the first element
389  auto it = chosen.begin();
390  std::advance(it, idx);
391  int chosen_agent = *it;
392 
393  // choose a random time from its path
394  int tmax = sol.paths[chosen_agent].back().interval.t_min - 1;
395  std::uniform_int_distribution<int> dist_time(0, tmax);
396  int chosen_t = dist_time(rnd_generator);
397 
398  // choose location
399  int rw_start_location = location_at_time(sol.paths[chosen_agent], chosen_t);
400 
401  // perform randomwalk
402  if (random_walk(chosen_agent, settings.destroy_settings.size, rw_start_location, chosen_t, upperbound, chosen))
403  {
404  break;
405  }
406  }
407 
408  // if not enough agents selected, fill with random agents
409  // find all agents that were not selected yet
410  std::vector<int> not_chosen;
411  for (int i = 0; i < instance.get_num_of_agents(); i++)
412  {
413  if (chosen.find(i) == chosen.end())
414  {
415  not_chosen.push_back(i);
416  }
417  }
418  int iterations = settings.destroy_settings.size - static_cast<int>(chosen.size());
419  for (int i = 0; i < iterations; i++)
420  {
421  // choose a random element that was not yet chosen
422  std::uniform_int_distribution<int> dist(0, not_chosen.size() - 1);
423  int idx = dist(rnd_generator);
424  auto it = not_chosen.begin();
425  std::advance(it, idx);
426  chosen.insert(*it); // add to the neighborhood
427  not_chosen.erase(it); // remove from the not yet chosen elements
428  }
429  assertm(static_cast<int>(chosen.size()) == settings.destroy_settings.size, "Neighborhood is not full.");
430  sol.destroyed_paths = std::vector(std::make_move_iterator(chosen.begin()), std::make_move_iterator(chosen.end()));
431  // shuffle the paths
432  std::shuffle(sol.destroyed_paths.begin(), sol.destroyed_paths.end(), rnd_generator);
433  assertm(sol.destroyed_paths.size() > 0, "No paths were destroyed.");
434  sol.feasible = false;
435 }

438 auto LNS::random_walk(int agent_num, int neighborhood_size, int start_location, int start_time, int upperbound,
439  std::unordered_set<int>& chosen) const -> bool
440 {
441  int curr = start_location;
442  for (int t = start_time; t < upperbound; t++)
443  {
444  std::vector<int> next_locations = instance.get_neighbor_locations(curr);
445  next_locations.push_back(curr); // agent can also stay
446  std::shuffle(next_locations.begin(), next_locations.end(), rnd_generator);
447  // try all possible neighbors in random order
448  for (auto loc : next_locations)
449  {
450  int next_h_val = instance.get_heuristic_distance(agent_num, loc);
451  // if lowerbound on the time to reach goal smaller than limit, accept the move
452  if (t + 1 + next_h_val < upperbound)
453  {
454  // get the agent that is in conflict and add it to the neighborhood
455  int conflicting_agent = constraint_table.get_blocking_agent(curr, loc, t + 1);
456 
457  // if there is an agent in collision, add it to the neighborhood
458  if (conflicting_agent != -1)
459  {
460  chosen.insert(conflicting_agent);
461  }
462 
463  curr = loc;
464  break;
465  }
466  }
467  // test whether enough agents selected
468  if (static_cast<int>(chosen.size()) == neighborhood_size)
469  {
470  return true;
471  }
472  }
473  assertm(static_cast<int>(chosen.size()) <= neighborhood_size, "Enough agents selected, but false returned.");
474  return false;
475 }

477 void LNS::destroy_intersection(Solution& sol) const
478 {
479  last_destroy_strategy = DESTROY_TYPE::INTERSECTION;
480  assertm(constraint_table_initialized, "Constraint table is not initialized.");
481 
482  // find all intersection vertices
483  std::vector<int> intersection_vertices_free;
484  intersection_vertices_free.reserve(instance.get_num_free_cells()); // upperbound on number of intersection vertices
485 
486  for (int i = 0; i < instance.get_num_free_cells(); i++)
487  {
488  const auto& agents_counts = constraint_table.get_agents_counts_free(i);
489  if (static_cast<int>(agents_counts.size()) >= 3)
490  {
491  intersection_vertices_free.push_back(i);
492  }
493  }
494 
495  // create queue
496  std::queue<int> openlist;
497 
498  // pick a random intersection vertex
499  {
500  std::uniform_int_distribution<int> dist(0, intersection_vertices_free.size() - 1);
501  int idx = dist(rnd_generator);
502  int chosen = instance.free_location_to_location(intersection_vertices_free[idx]);
503  openlist.push(chosen);
504  }
505 
506 
507  std::unordered_set<int> neighborhood; // TODO try set
508  std::vector<bool> known(instance.get_num_cells(), false);
509 
510  while (!openlist.empty())
511  {
512  // retrieve current node
513  int current = openlist.front();
514  openlist.pop();
515  // std::cout << "Current: " << current << std::endl;
516  if (known[current])
517  {
518  continue;
519  }
520  known[current] = true;
521 
522  // check whether intersection vertex
523  auto agents_counts = constraint_table.get_agents_counts(current);
524  if (static_cast<int>(agents_counts.size()) >= 3)
525  {
526  // add agents that visit this vertex to the neighborhood
527  get_intersection_agents(neighborhood, current);
528  }
529 
530  // check whether enough agents selected
531  // std::cout << "Selected: " << static_cast<int>(neighborhood.size()) << std::endl;
532  if (settings.destroy_settings.size == static_cast<int>(neighborhood.size()))
533  {
534  break;
535  }
536 
537  // add neighbors to the queue
538  auto neighbors = instance.get_neighbor_locations(current);
539  for (auto it : neighbors)
540  {
541  openlist.push(it);
542  }
543  }
544 
545  // if not enough agents selected, fill with random agents
546  // find all agents that were not selected yet
547  std::vector<int> not_chosen;
548  for (int i = 0; i < instance.get_num_of_agents(); i++)
549  {
550  if (neighborhood.find(i) == neighborhood.end())
551  {
552  not_chosen.push_back(i);
553  }
554  }
555  int iterations = settings.destroy_settings.size - static_cast<int>(neighborhood.size());
556  for (int i = 0; i < iterations; i++)
557  {
558  // choose a random element that was not yet chosen
559  std::uniform_int_distribution<int> dist(0, not_chosen.size() - 1);
560  int idx = dist(rnd_generator);
561  auto it = not_chosen.begin();
562  std::advance(it, idx);
563  neighborhood.insert(*it); // add to the neighborhood
564  not_chosen.erase(it); // remove from the not yet chosen elements
565  }
566 
567  assertm(settings.destroy_settings.size == static_cast<int>(neighborhood.size()), "Wrong neighborhood size.");
568  // convert to vector and shuffle
569  sol.destroyed_paths = std::vector(std::make_move_iterator(neighborhood.begin()), std::make_move_iterator(neighborhood.end()));
570  // shuffle the paths
571  std::shuffle(sol.destroyed_paths.begin(), sol.destroyed_paths.end(), rnd_generator);
572  assertm(sol.destroyed_paths.size() > 0, "No paths were destroyed.");
573  sol.feasible = false;
574 }

576 void LNS::get_intersection_agents(std::unordered_set<int>& neighborhood, int current) const
577 {
578  // find maximal T
579  int t_max = constraint_table.get_last_constraint_start(current);
580 
581  // choose random time
582  std::uniform_int_distribution<int> dist(0, t_max);
583  int t = dist(rnd_generator);
584  int delta = 0;
585  while (static_cast<int>(neighborhood.size()) < settings.destroy_settings.size && t + delta <= t_max && t - delta >= 0)
586  {
587  // find the agent, that blocks the cell at time t + delta
588  int blocking_agent_1 = constraint_table.get_blocking_agent(current, current, t + delta);
589  if (blocking_agent_1 >= 0)
590  {
591  neighborhood.insert(blocking_agent_1);
592  }
593  // find the agent, that blocks the cell at time t - delta
594  int blocking_agent_2 = constraint_table.get_blocking_agent(current, current, t - delta);
595  if (static_cast<int>(neighborhood.size()) < settings.destroy_settings.size && blocking_agent_2 >= 0)
596  {
597  neighborhood.insert(blocking_agent_2);
598  }
599  delta++;
600  }
601 }

603 void LNS::destroy_adaptive(Solution& sol) const
604 {
605  // choose strategy by roulette wheel selection
606  // calculate the sum of weights
607  auto sum_of_weights = std::reduce(destroy_weights.begin(), destroy_weights.end());
608  assertm(sum_of_weights > 0, "Invalid weights.");
609 
610  // choose a random number in the range [0, sum_of_weights]
611  std::uniform_real_distribution<double> dist(0.0, sum_of_weights);
612  double rand_value = dist(rnd_generator);
613 
614  // spin the roulette
615  double cummulative_sum = 0.0;
616  DESTROY_TYPE chosen_type = DESTROY_TYPE::ADAPTIVE;
617  for (int i = 0; i < static_cast<int>(destroy_weights.size()); i++)
618  {
619  cummulative_sum += destroy_weights[i];
620  if (rand_value <= cummulative_sum)
621  {
622  auto chosen = magic_enum::enum_cast<DESTROY_TYPE>(i);
623  if (chosen.has_value())
624  {
625  chosen_type = chosen.value();
626  break;
627  }
628  throw std::runtime_error("Unknown DESTROY_TYPE encountered in adaptive destroy.");
629  }
630  }

679 void LNS::repair_default(Solution& sol) const
680 {
681  assertm(!sol.feasible, "Can not repair a feasible solution");
682  assertm((int)sol.destroyed_paths.size() > 0, "There are no destroyed paths in the solution to be repaired.");
683 
684  if (settings.sipp_settings.info_type == INFO_type::visualisation)
685  {
686  sipp_info.clear();
687  sipp_info.resize(sol.destroyed_paths.size());
688  }
689 
690  // remove the destroyed paths
691  for (auto& it : sol.destroyed_paths)
692  {
693  // std::cout << "Removing path: " << it << std::endl;
694  planner->safe_interval_table.remove_constraints(sol.paths[it]);
695  if (constraint_table_initialized)
696  {
697  constraint_table.remove_constraints(sol.paths[it], it);
698  }
699  sol.paths[it].clear();
700  already_planned.erase(it);
701  }
702  sol.feasible = true;
703 
704  // replan
705  for (int i = 0; i < static_cast<int>(sol.destroyed_paths.size()); i++)
706  {
707  int path_idx = sol.destroyed_paths[i];
708  TimePointPath tp_path = planner->plan(path_idx, already_planned);
709  if (tp_path.empty())
710  {
711  // TODO reset the safe interval table and constrant table (is done in discard solution)
712  sol.feasible = false;
713  return;
714  }
715 
716  // get the visualization info
717  if (settings.sipp_settings.info_type == INFO_type::visualisation)
718  {
719  sipp_info[i] = std::move(planner->iter_info);
720  planner->iter_info.clear();
721  }
722 
723  planner->safe_interval_table.add_constraints(tp_path);
724  if (constraint_table_initialized)
725  {
726  constraint_table.add_constraints(tp_path, path_idx);
727  }
728  sol.paths[path_idx] = tp_path;
729  }
