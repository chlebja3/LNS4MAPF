72 std::vector<Point2d> Map::find_neighbors(const Point2d& point) const
73 {
74  if (!is_in(point)) return {}; // invalid point
75  std::vector<Point2d> ret;
76  const std::array<Point2d, 4> neighbors = {Point2d(0, 1), Point2d(1, 0), Point2d(0, -1), Point2d(-1, 0)};
77  for (auto& it : neighbors)
78  {
79  Point2d neigh = point + it;
80  if (!is_in(neigh)) continue;
81  if (index(neigh) != 0) continue;
82  ret.push_back(neigh);
83  }
84  return ret;
85 }
