51 void Instance::load_map(void)
52 {
53  int num_of_cols, num_of_rows = 0;
54  // open the map file
55  std::ifstream map_file(map_fname.c_str());
56 
57  // check if the open was successful
58  if (!map_file.is_open()) throw std::runtime_error("Could not open map file.");
59 
60  // read the file line by line
61  std::string line;
62  while (getline(map_file, line))
63  {
64  // split the line
65  std::vector<std::string> split_line;
66  boost::split(split_line, line, boost::is_any_of(" "));
67 
68  // filter out invalid lines
69  if (split_line.size() == 0) continue;
70 
71  if (boost::iequals(split_line[0], "type"))
72  type = split_line[1]; // read type
73  else if (boost::iequals(split_line[0], "height")) // read height
74  {
75  try
76  {
77  num_of_rows = stoi(split_line[1]);
78  }
79  catch (std::exception& e)
80  {
81  throw std::runtime_error("Unable to read the number of rows.");
82  }
83  }
84  else if (boost::iequals(split_line[0], "width")) // read width
85  {
86  try
87  {
88  num_of_cols = stoi(split_line[1]);
89  }
90  catch (std::exception& e)
91  {
92  throw std::runtime_error("Unable to read the number of rows.");
93  }
94  }
95  else if (boost::iequals(split_line[0], "map")) // read the map
96  {
97  while (getline(map_file, line))
98  {
99  for (int i = 0; i < (int)line.size(); i++)
100  {
101  // treat . as free cell
102  if (line[i] == '.') map_data.data.push_back(0);
103  // treat @, T as obstacle
104  else if (line[i] == '@' || line[i] == 'T')
105  {
106  map_data.data.push_back(1);
107  }
108  // report all other symbols as unknown
109  else
110  {
111  throw std::runtime_error("Unknown symbol in map description.");
112  }
113  }
114  // check whether the number of cells in this row is consistent with num_of_columns
115  if (map_data.data.size() % num_of_cols != 0)
116  { throw std::runtime_error("Inconsistent number of symbols in a row in the map description."); }
117  }
118  // check whether the number of lines is consistent with num_of_rows
119  if ((int)map_data.data.size() != num_of_cols * num_of_rows)
120  { throw std::runtime_error("Inconsistent number of rows in the map description."); }
121  }
122  }
123 
124  if (num_of_cols <= 0) throw std::runtime_error("Invalid number of columns.");
125  if (num_of_rows <= 0) throw std::runtime_error("Invalid number of rows.");
126 
127  // close the map file
128  map_file.close();
129  map_data.width = num_of_cols;
130  map_data.height = num_of_rows;
131 }

133 void Instance::load_scene(void)
134 {
135  // open the scene file
136  std::ifstream scene_file(scene_fname.c_str());
137 
138  // check if the open was successful
139  if (!scene_file.is_open()) throw std::runtime_error("Unable to open scene file.");
140 
141  std::string line;
142 
143  // read the version
144  if (!getline(scene_file, line) || line != "version 1") throw std::invalid_argument("Invalid version of scene file.");
145 
146  int loaded_agents = 0;
147  start_positions.reserve(num_of_agents);
148  goal_positions.reserve(num_of_agents);
149  // read the file line by line until enough agents were loaded
150  while (loaded_agents < num_of_agents && getline(scene_file, line))
151  {
152  // split the line - the format is 'bucket, map, map_width, map_height, start_x, start_y, goal_x, goal_y, optimal_length'
153  std::vector<std::string> split_line;
154  boost::split(split_line, line, boost::is_any_of("\t "));
155 
156  // filter out invalid lines
157  if (split_line.size() != 9)
158  {
159  std::cout << "Invalid line in scen file: " << line << " length: " << split_line.size() << std::endl;
160  continue;
161  }
162 
163  // read the start and goal positions
164  try
165  {
166  const int start_x = stoi(split_line[4]);
167  const int start_y = stoi(split_line[5]);
168  const int goal_x = stoi(split_line[6]);
169  const int goal_y = stoi(split_line[7]);
170 
171  // store the positions in the list
172  start_positions.emplace_back(start_x, start_y);
173  goal_positions.emplace_back(goal_x, goal_y);
174  loaded_agents++;
175  }
176  catch (std::exception& e)
177  {
178  throw std::runtime_error("Could not read start and goal positions: '" + (std::string)e.what() + "'");
179  }
180  }
181 
182  // cloase the map file
183  scene_file.close();
184 
185  // check whether enough agents were loaded
186  if (loaded_agents != num_of_agents)
187  {
188  std::cout << "WARNING: Unable to load " << num_of_agents << " agents, as only " << loaded_agents
189  << " agents are available in the scene file." << std::endl;
190  num_of_agents = loaded_agents;
191  }
192 }

223 void Instance::calculate_optimal_paths(void)
224 {
225  optimal_paths.clear();
226  optimal_paths.reserve(num_of_agents);
227  // calculate the optimal paths one by one
228  for (int i = 0; i < num_of_agents; i++)

236 void Instance::calculate_optimal_paths_parallel(void)
237 {
238  optimal_paths.clear();
239  optimal_paths.reserve(num_of_agents);
240  // calculate the optimal paths one by one
241 
242  // prepare the vector with future paths
243  std::vector<std::future<Path>> futures;
244 
245  // calculate asynchronously the paths

247  {
248  futures.push_back(

253  for (int i = 0; i < num_of_agents; i++) { optimal_paths.push_back(futures[i].get()); }
