32  // open the map file
33  std::ifstream map_file(map_fname.c_str());
34 
35  // check if the open was successful
36  if (!map_file.is_open())
37  {
38  return false;
39  }
40 
41  // read the file line by line
42  std::string line;
43  while (getline(map_file, line))
44  {
45  // split the line
46  std::vector<std::string> split_line;
47  boost::split(split_line, line, boost::is_any_of(" "));
48 
49  // filter out invalid lines
50  if (split_line.size() == 0)
51  continue;
52 
53  if (boost::iequals(split_line[0], "type"))
54  type = split_line[1]; // read type
55  else if (boost::iequals(split_line[0], "height")) // read height
56  {
57  try
58  {
59  num_of_rows = stoi(split_line[1]);
60  }
61  catch (std::exception& e)
62  {
63  std::cout << "Could not read height: '" << e.what() << "'" << std::endl;
64  }
65  }
66  else if (boost::iequals(split_line[0], "width")) // read width
67  {
68  try
69  {
70  num_of_cols = stoi(split_line[1]);
71  }
72  catch (std::exception& e)
73  {
74  std::cout << "Could not read width: '" << e.what() << "'" << std::endl;
75  }
76  }
77  else if (boost::iequals(split_line[0], "map")) // read the map
78  {
79  while (getline(map_file, line))
80  {
81  for (int i = 0; i < (int)line.size(); i++)
82  {
83  // treat . as free cell
84  if (line[i] == '.')
85  {
86  map_data.push_back(0);
87  }
88  // treat @, T as obstacle
89  else if (line[i] == '@' || line[i] == 'T')
90  {
91  map_data.push_back(1);
92  }
93  // report all other symbols as unknown
94  else
95  {
96  throw std::runtime_error("Unknown symbol in map description.");
97  }
98  }
99  // check whether the number of cells in this row is consistent with num_of_columns
100  if (map_data.size() % num_of_cols != 0)
101  {
102  throw std::runtime_error("Inconsistent number of symbols in a row in the map description.");
103  }
104  }
105  // check whether the number of lines is consistent with num_of_rows
106  if ((int)map_data.size() != num_of_cols * num_of_rows)
107  {
108  throw std::runtime_error("Inconsistent number of rows in the map description.");
109  }
110  }
111  }
112 
113  // cloase the map file
114  map_file.close();

130  std::string line;
131 
132  // read the version

135 
136  int loaded_agents = 0;
137  start_positions.reserve(num_of_agents);
138  goal_positions.reserve(num_of_agents);
139  // read the file line by line until enough agents were loaded
140  while (loaded_agents < num_of_agents && getline(scene_file, line))
141  {
142  // split the line - the format is 'bucket, map, map_width, map_height, start_x, start_y, goal_x, goal_y, optimal_length'
143  std::vector<std::string> split_line;
144  boost::split(split_line, line, boost::is_any_of("\t"));
145 
146  // filter out invalid lines
147  if (split_line.size() != 9)
148  continue;
149 
150  // read the start and goal positions
151  try
152  {
153  int start_x = stoi(split_line[4]);
154  int start_y = stoi(split_line[5]);
155  int goal_x = stoi(split_line[6]);
156  int goal_y = stoi(split_line[7]);
157 
158  // store the positions in the list
159  start_positions.push_back(std::make_tuple(start_x, start_y));
160  goal_positions.push_back(std::make_tuple(goal_x, goal_y));
161  loaded_agents++;
162  }
163  catch (std::exception& e)
164  {
165  std::cout << "Could not read start and goal positions: '" << e.what() << "'" << std::endl;
166  }
167  }
168 
169  // cloase the map file
170  scene_file.close();
171 
172  // check whether enough agents were loaded
173  if (loaded_agents != num_of_agents)
174  {
175  std::cout << "WARNING: Unable to load " << num_of_agents << " agents, as only " << loaded_agents << " agents are available in the scene file."
176  << std::endl;

178  }
179 
180  return true;
