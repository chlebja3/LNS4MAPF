241 // Render the map onto a texture
242 void Visualizer::mapToTexture(const Map& map_data)
243 {
244  // initialize the texture
245  sf::RenderTexture map_render_texture;
246  map_render_texture.create(map_data.width * CELL_SIZE, map_data.height * CELL_SIZE);
247 
248  // clear the texture with a background color
249  map_render_texture.clear(sf::Color::White);
250 
251  // Draw the grid onto the texture
252  for (int y = 0; y < map_data.height; y++)
253  {
254  for (int x = 0; x < map_data.width; x++)
255  {
256  // create the cell
257  sf::RectangleShape cell(sf::Vector2f(CELL_SIZE, CELL_SIZE));
258  cell.setPosition((float)x * CELL_SIZE, (float)y * CELL_SIZE);
259  if ((int)map_data.data[y * map_data.width + x] == 1)
260  {
261  // obstacles
262  cell.setFillColor(sf::Color::Black);
263  }
264  else
265  {
266  // free cells
267  cell.setFillColor(sf::Color::White);
268  }
269 
270  // grid color and thickness
271  cell.setOutlineThickness(-BORDER_SIZE);
272  cell.setOutlineColor(sf::Color::Black);
273 
274  map_render_texture.draw(cell);
275  }
276  }
277 
278  // Finalize the rendering
279  map_render_texture.display();
280 
281  // set the map texture
282  map_texture = map_render_texture.getTexture();
283  // turn on antialiasing
284  // map_texture.setSmooth(true);
285  map_sprite = sf::Sprite(map_texture);
286  scale = (double)WINDOW_WIDTH / (CELL_SIZE * map_data.width);
287  map_sprite.setScale(sf::Vector2f(scale, scale));
288 }

330 void Visualizer::draw_arrow(sf::RenderWindow& window, const Point2d& start, const Point2d& end, const sf::Color& color)
331 {
332  const double x = (end.x + start.x) / 2.;
333  const double y = (end.y + start.y) / 2.;
334 
335  // draw the line
336  sf::Vector2f shape;
337  sf::RectangleShape line;
338  if (start.x == end.x) // vertical
339  {
340  shape = sf::Vector2f(CELL_SIZE / (float)LINE_WIDTH, CELL_SIZE + (float)CELL_SIZE / LINE_WIDTH);
341  line = sf::RectangleShape(shape);
342  line.setPosition((x + (float)1 / LINE_WIDTH) * scale * CELL_SIZE, (y - (float)1 / (2 * LINE_WIDTH)) * scale * CELL_SIZE);
343  }
344  else // horizontal
345  {
346  shape = sf::Vector2f(CELL_SIZE + (float)CELL_SIZE / LINE_WIDTH, CELL_SIZE / (float)LINE_WIDTH);
347  line = sf::RectangleShape(shape);
348  line.setPosition((x - (float)1 / (2 * LINE_WIDTH)) * scale * CELL_SIZE, (y + (float)1 / LINE_WIDTH) * scale * CELL_SIZE);
349  }
350  line.setScale(sf::Vector2f(scale, scale));
351  line.setFillColor(color);
352  window.draw(line);
