67 void Visualizer::run()
68 {
69  int window_height = std::round(WINDOW_WIDTH * (map_height / map_width));
70 
71  // turn on antialiasing
72  sf::ContextSettings settings;
73  settings.antialiasingLevel = 8.0;
74 
75  sf::RenderWindow window(sf::VideoMode(WINDOW_WIDTH, window_height), "Visualization", sf::Style::Default, settings);
76 
77  // set framerate
78  window.setFramerateLimit(60);
79 
80  // Activate the window for OpenGL rendering
81  window.setActive();
82 
83  // create a view
84  sf::View view(sf::FloatRect(0, 0, WINDOW_WIDTH, window_height));
85 
86  // Zoom settings
87  float zoomLevel = 1.0f;
88  const float zoomStep = 1.1f;
89 
90  while (window.isOpen() && running)
91  {
92  // poll all events from the window
93  sf::Event event;
94  while (window.pollEvent(event))
95  {
96  // close event
97  if (event.type == sf::Event::Closed)
98  {
99  running = false;
100  }
101  // zoom event
102  else if (event.type == sf::Event::MouseWheelScrolled)
103  {
104  if (event.mouseWheelScroll.delta > 0)
105  {
106  view.zoom(1 / zoomStep); // Zoom in
107  zoomLevel *= 1 / zoomStep;
108  }
109  else
110  {
111  view.zoom(zoomStep); // Zoom out
112  zoomLevel *= zoomStep;
113  }
114  }
115  // Panning with arrow keys
116  else if (sf::Keyboard::isKeyPressed(sf::Keyboard::Left))
117  {
118  view.move(-10 * zoomLevel, 0);
119  }
120  else if (sf::Keyboard::isKeyPressed(sf::Keyboard::Right))
121  {
122  view.move(10 * zoomLevel, 0);
123  }
124  else if (sf::Keyboard::isKeyPressed(sf::Keyboard::Up))
125  {
126  view.move(0, -10 * zoomLevel);
127  }
128  else if (sf::Keyboard::isKeyPressed(sf::Keyboard::Down))
129  {
130  view.move(0, 10 * zoomLevel);
131  }
132  }
133 
134  // clear the window
135  window.clear(sf::Color::Black);
136 
137  // draw the map
138  window.draw(map_sprite);
139 
140  // draw the agents
141  draw_agents(window);
142  draw_goals(window);
143 
144  // set the view
145  window.setView(view);
146 
147  // display
148  window.display();
149  }
150 
151  // close the window
152  if (window.isOpen())
153  window.close();
154  std::cout << "Visualizer thread ended" << std::endl;
155 }

158 void Visualizer::mapToTexture(const std::vector<std::uint8_t>& map_data, int num_of_rows, int num_of_columns)
159 {
160  // initialize the texture
161  sf::RenderTexture map_render_texture;
162  map_render_texture.create(num_of_columns * CELL_SIZE, num_of_rows * CELL_SIZE);
163 
164  // clear the texture with a background color
165  map_render_texture.clear(sf::Color::White);
166 
167  // Draw the grid onto the texture
168  for (int y = 0; y < num_of_rows; y++)
169  {
170  for (int x = 0; x < num_of_columns; x++)
171  {
172  // create the cell
173  sf::RectangleShape cell(sf::Vector2f(CELL_SIZE, CELL_SIZE));
174  cell.setPosition(x * CELL_SIZE, y * CELL_SIZE);
175  if ((int)map_data[y * num_of_columns + x] == 1)
176  {
177  // obstacles
178  cell.setFillColor(sf::Color::Black);
179  }
180  else
181  {
182  // free cells
183  cell.setFillColor(sf::Color::White);
184  }
185 
186  // grid color and thickness
187  cell.setOutlineThickness(-BORDER_SIZE);
188  cell.setOutlineColor(sf::Color::Black);
189 
190  map_render_texture.draw(cell);
191  }
192  }
193 
194  // Finalize the rendering
195  map_render_texture.display();
196 
197  // set the map texture
198  map_texture = map_render_texture.getTexture();
199  // turn on antialiasing
200  // map_texture.setSmooth(true);
201  map_sprite = sf::Sprite(map_texture);
202  scale = (double)WINDOW_WIDTH / (CELL_SIZE * num_of_columns);
203  map_sprite.setScale(sf::Vector2f(scale, scale));
204 }

206 void Visualizer::draw_agents(sf::RenderWindow& window)

219  agents_circles.back().setScale(sf::Vector2f(scale, scale));
